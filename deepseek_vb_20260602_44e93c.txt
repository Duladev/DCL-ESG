Imports System.Drawing
Imports System.Drawing.Drawing2D
Imports System.Windows.Forms

''' <summary>
''' Centralized ESG Theme — Fully tested working version
''' Usage: AppTheme.Apply(Me) in Form_Load
''' </summary>
Public Module AppTheme

    ' ─── Colour Palette (Nature / Sustainability) ────────────────────────────────
    Public ReadOnly ColorBackground As Color = Color.FromArgb(248, 251, 245)     ' #f8fbf5
    Public ReadOnly ColorSurface As Color = Color.FromArgb(255, 255, 255)        
    Public ReadOnly ColorCard As Color = Color.FromArgb(230, 255, 191)           ' #e6ffbf
    Public ReadOnly ColorPrimary As Color = Color.FromArgb(33, 86, 0)            ' #215600
    Public ReadOnly ColorPrimaryDark As Color = Color.FromArgb(25, 65, 0)        
    Public ReadOnly ColorPrimaryLight As Color = Color.FromArgb(80, 160, 30)     
    Public ReadOnly ColorAccent As Color = Color.FromArgb(100, 180, 50)          
    Public ReadOnly ColorText As Color = Color.FromArgb(33, 86, 0)               
    Public ReadOnly ColorTextMuted As Color = Color.FromArgb(100, 120, 90)       
    Public ReadOnly ColorBorder As Color = Color.FromArgb(180, 200, 150)         
    Public ReadOnly ColorDanger As Color = Color.FromArgb(220, 60, 50)           
    Public ReadOnly ColorWarning As Color = Color.FromArgb(245, 158, 11)         

    ' ─── Typography ───────────────────────────────────────────────────────────
    Public ReadOnly FontTitle As New Font("Segoe UI", 18, FontStyle.Bold)
    Public ReadOnly FontSubtitle As New Font("Segoe UI", 11, FontStyle.Regular)
    Public ReadOnly FontButton As New Font("Segoe UI Semibold", 9, FontStyle.Bold)
    Public ReadOnly FontBody As New Font("Segoe UI", 9, FontStyle.Regular)
    Public ReadOnly FontSmall As New Font("Segoe UI", 8, FontStyle.Regular)

    ' ─── Sizing ───────────────────────────────────────────────────────────────
    Public Const ButtonRadius As Integer = 20
    Public Const CardRadius As Integer = 12

    ' ─── Apply Theme to a Form ────────────────────────────────────────────────
    Public Sub Apply(frm As Form, Optional backgroundImage As Image = Nothing, Optional useBlur As Boolean = False)
        frm.BackColor = ColorBackground
        frm.ForeColor = ColorText
        frm.Font = FontBody
        frm.FormBorderStyle = FormBorderStyle.None

        ' Set background if provided
        If backgroundImage IsNot Nothing Then
            frm.BackgroundImage = backgroundImage
            frm.BackgroundImageLayout = ImageLayout.Stretch
        End If

        ' Round the form corners
        SetRoundRegion(frm, CardRadius)

        ' Style all child controls
        StyleAllControls(frm)

        ' Keep rounded corners on resize
        AddHandler frm.Resize, Sub(s, e) SetRoundRegion(frm, CardRadius)
    End Sub

    ' ─── Style all controls recursively ───────────────────────────────────────
    Private Sub StyleAllControls(parent As Control)
        For Each ctrl As Control In parent.Controls
            ' Style based on control type
            Select Case True
                Case TypeOf ctrl Is Button
                    StyleButton(DirectCast(ctrl, Button))
                Case TypeOf ctrl Is Label
                    StyleLabel(DirectCast(ctrl, Label))
                Case TypeOf ctrl Is TextBox
                    StyleTextBox(DirectCast(ctrl, TextBox))
                Case TypeOf ctrl Is ComboBox
                    StyleComboBox(DirectCast(ctrl, ComboBox))
                Case TypeOf ctrl Is DataGridView
                    StyleGridView(DirectCast(ctrl, DataGridView))
                Case TypeOf ctrl Is Panel
                    StylePanel(DirectCast(ctrl, Panel))
                Case TypeOf ctrl Is GroupBox
                    StyleGroupBox(DirectCast(ctrl, GroupBox))
                Case TypeOf ctrl Is TabControl
                    StyleTabControl(DirectCast(ctrl, TabControl))
                Case TypeOf ctrl Is RichTextBox
                    StyleRichTextBox(DirectCast(ctrl, RichTextBox))
                Case TypeOf ctrl Is ListBox
                    StyleListBox(DirectCast(ctrl, ListBox))
                Case TypeOf ctrl Is DateTimePicker
                    StyleDateTimePicker(DirectCast(ctrl, DateTimePicker))
                Case TypeOf ctrl Is NumericUpDown
                    StyleNumericUpDown(DirectCast(ctrl, NumericUpDown))
            End Select

            ' Recursively style child controls
            If ctrl.HasChildren Then
                StyleAllControls(ctrl)
            End If
        Next
    End Sub

    ' ─── BUTTON Styling ───────────────────────────────────────────────────────
    Private Sub StyleButton(btn As Button, Optional style As ButtonStyle = ButtonStyle.Primary)
        btn.FlatStyle = FlatStyle.Flat
        btn.FlatAppearance.BorderSize = 0
        btn.Font = FontButton
        btn.ForeColor = Color.White
        btn.Cursor = Cursors.Hand
        btn.TextAlign = ContentAlignment.MiddleCenter
        btn.BackColor = GetButtonColor(style)
        btn.Tag = style

        ' Custom painting for rounded corners
        AddHandler btn.Paint, Sub(sender, e)
                                  Dim b = DirectCast(sender, Button)
                                  Dim g = e.Graphics
                                  g.SmoothingMode = SmoothingMode.AntiAlias
                                  
                                  Dim rect = New Rectangle(0, 0, b.Width - 1, b.Height - 1)
                                  Dim radius = Math.Min(ButtonRadius, Math.Min(b.Height, b.Width) \ 2)
                                  Using path = RoundedRect(rect, radius)
                                      Using brush As New SolidBrush(b.BackColor)
                                          g.FillPath(brush, path)
                                      End Using
                                      Using pen As New Pen(Color.FromArgb(80, Color.White), 1)
                                          g.DrawPath(pen, path)
                                      End Using
                                  End Using
                                  
                                  TextRenderer.DrawText(g, b.Text, b.Font, rect, b.ForeColor, 
                                                       TextFormatFlags.HorizontalCenter Or TextFormatFlags.VerticalCenter)
                              End Sub

        ' Hover effects
        AddHandler btn.MouseEnter, Sub(sender, e) 
                                       Dim b = DirectCast(sender, Button)
                                       b.BackColor = AdjustBrightness(b.BackColor, 1.1)
                                       b.Invalidate()
                                   End Sub
        AddHandler btn.MouseLeave, Sub(sender, e) 
                                       Dim b = DirectCast(sender, Button)
                                       Dim originalStyle = If(TypeOf b.Tag Is ButtonStyle, DirectCast(b.Tag, ButtonStyle), ButtonStyle.Primary)
                                       b.BackColor = GetButtonColor(originalStyle)
                                       b.Invalidate()
                                   End Sub
        AddHandler btn.MouseDown, Sub(sender, e) 
                                      Dim b = DirectCast(sender, Button)
                                      b.BackColor = AdjustBrightness(b.BackColor, 0.9)
                                      b.Invalidate()
                                  End Sub
        AddHandler btn.MouseUp, Sub(sender, e) 
                                    Dim b = DirectCast(sender, Button)
                                    b.BackColor = AdjustBrightness(b.BackColor, 1.05)
                                    b.Invalidate()
                                End Sub
    End Sub

    Private Function GetButtonColor(style As ButtonStyle) As Color
        Select Case style
            Case ButtonStyle.Primary : Return ColorPrimary
            Case ButtonStyle.Secondary : Return ColorCard
            Case ButtonStyle.Danger : Return ColorDanger
            Case ButtonStyle.Accent : Return ColorAccent
            Case Else : Return ColorPrimary
        End Select
    End Function

    ' ─── LABEL Styling ────────────────────────────────────────────────────────
    Private Sub StyleLabel(lbl As Label)
        lbl.BackColor = Color.Transparent
        lbl.ForeColor = ColorText
    End Sub

    ' ─── TEXTBOX Styling (Clean & Simple) ─────────────────────────────────────
    Private Sub StyleTextBox(txt As TextBox)
        txt.BorderStyle = BorderStyle.FixedSingle
        txt.BackColor = Color.White
        txt.ForeColor = ColorText
        txt.Font = FontBody
    End Sub

    ' ─── COMBOBOX Styling ─────────────────────────────────────────────────────
    Private Sub StyleComboBox(cmb As ComboBox)
        cmb.FlatStyle = FlatStyle.Flat
        cmb.BackColor = Color.White
        cmb.ForeColor = ColorText
        cmb.Font = FontBody
    End Sub

    ' ─── RICH TEXTBOX Styling ─────────────────────────────────────────────────
    Private Sub StyleRichTextBox(rtb As RichTextBox)
        rtb.BorderStyle = BorderStyle.FixedSingle
        rtb.BackColor = Color.White
        rtb.ForeColor = ColorText
        rtb.Font = FontBody
    End Sub

    ' ─── LISTBOX Styling ──────────────────────────────────────────────────────
    Private Sub StyleListBox(lst As ListBox)
        lst.BorderStyle = BorderStyle.FixedSingle
        lst.BackColor = Color.White
        lst.ForeColor = ColorText
        lst.Font = FontBody
    End Sub

    ' ─── DATETIMEPICKER Styling ───────────────────────────────────────────────
    Private Sub StyleDateTimePicker(dtp As DateTimePicker)
        dtp.Font = FontBody
        dtp.CalendarForeColor = ColorText
        dtp.CalendarTitleBackColor = ColorPrimary
        dtp.CalendarTitleForeColor = Color.White
        dtp.CalendarMonthBackground = Color.White
    End Sub

    ' ─── NUMERICUPDOWN Styling ─────────────────────────────────────────────────
    Private Sub StyleNumericUpDown(num As NumericUpDown)
        num.BorderStyle = BorderStyle.FixedSingle
        num.BackColor = Color.White
        num.ForeColor = ColorText
        num.Font = FontBody
    End Sub

    ' ─── DATAGRIDVIEW Styling ─────────────────────────────────────────────────
    Private Sub StyleGridView(dgv As DataGridView)
        dgv.BackgroundColor = Color.White
        dgv.GridColor = ColorBorder
        dgv.BorderStyle = BorderStyle.None
        dgv.CellBorderStyle = DataGridViewCellBorderStyle.SingleHorizontal
        dgv.ColumnHeadersBorderStyle = DataGridViewHeaderBorderStyle.None
        dgv.EnableHeadersVisualStyles = False
        dgv.RowHeadersVisible = False
        dgv.SelectionMode = DataGridViewSelectionMode.FullRowSelect
        dgv.AutoSizeRowsMode = DataGridViewAutoSizeRowsMode.None
        
        ' Default cell styles
        dgv.DefaultCellStyle.BackColor = Color.White
        dgv.DefaultCellStyle.ForeColor = ColorText
        dgv.DefaultCellStyle.SelectionBackColor = ColorPrimary
        dgv.DefaultCellStyle.SelectionForeColor = Color.White
        dgv.DefaultCellStyle.Font = FontBody
        
        ' Alternating rows
        dgv.AlternatingRowsDefaultCellStyle.BackColor = Color.FromArgb(230, 255, 191)
        
        ' Column headers
        dgv.ColumnHeadersDefaultCellStyle.BackColor = ColorPrimary
        dgv.ColumnHeadersDefaultCellStyle.ForeColor = Color.White
        dgv.ColumnHeadersDefaultCellStyle.Font = FontButton
        dgv.ColumnHeadersHeight = 35
    End Sub

    ' ─── PANEL Styling (Rounded corners with background) ──────────────────────
    Private Sub StylePanel(pnl As Panel)
        pnl.BackColor = Color.Transparent
        
        ' Store original background to restore
        Dim originalBackColor = pnl.BackColor
        
        AddHandler pnl.Paint, Sub(sender, e)
                                  Dim p = DirectCast(sender, Panel)
                                  Dim g = e.Graphics
                                  g.SmoothingMode = SmoothingMode.AntiAlias
                                  
                                  Dim rect = New Rectangle(0, 0, p.Width - 1, p.Height - 1)
                                  Using path = RoundedRect(rect, CardRadius)
                                      Using brush As New SolidBrush(Color.FromArgb(220, ColorCard))
                                          g.FillPath(brush, path)
                                      End Using
                                      Using pen As New Pen(ColorBorder, 1)
                                          g.DrawPath(pen, path)
                                      End Using
                                  End Using
                              End Sub
        
        AddHandler pnl.Resize, Sub(sender, e) DirectCast(sender, Panel).Invalidate()
    End Sub

    ' ─── GROUPBOX Styling ─────────────────────────────────────────────────────
    Private Sub StyleGroupBox(grp As GroupBox)
        grp.BackColor = Color.Transparent
        grp.ForeColor = ColorPrimary
        grp.Font = FontButton
        
        AddHandler grp.Paint, Sub(sender, e)
                                  Dim g = DirectCast(sender, GroupBox)
                                  Dim graphics = e.Graphics
                                  graphics.SmoothingMode = SmoothingMode.AntiAlias
                                  
                                  Dim titleHeight = 20
                                  Dim rect = New Rectangle(0, titleHeight \ 2, g.Width - 1, g.Height - titleHeight \ 2 - 1)
                                  
                                  Using path = RoundedRect(rect, CardRadius)
                                      Using brush As New SolidBrush(Color.FromArgb(200, ColorCard))
                                          graphics.FillPath(brush, path)
                                      End Using
                                      Using pen As New Pen(ColorBorder, 1)
                                          graphics.DrawPath(pen, path)
                                      End Using
                                  End Using
                                  
                                  ' Draw title background
                                  Dim titleRect = New Rectangle(10, 0, g.Width - 20, titleHeight)
                                  Using brush As New SolidBrush(ColorCard)
                                      graphics.FillRectangle(brush, titleRect)
                                  End Using
                                  
                                  ' Draw title text
                                  Using brush As New SolidBrush(ColorPrimary)
                                      graphics.DrawString(g.Text, FontButton, brush, titleRect, 
                                                         New StringFormat With {.Alignment = StringAlignment.Center, 
                                                                               .LineAlignment = StringAlignment.Center})
                                  End Using
                              End Sub
        
        AddHandler grp.Resize, Sub(sender, e) DirectCast(sender, GroupBox).Invalidate()
    End Sub

    ' ─── TABCONTROL Styling ───────────────────────────────────────────────────
    Private Sub StyleTabControl(tab As TabControl)
        tab.DrawMode = TabDrawMode.OwnerDrawFixed
        tab.SizeMode = TabSizeMode.Fixed
        tab.ItemSize = New Size(120, 35)
        tab.Padding = New Point(10, 5)
        
        AddHandler tab.DrawItem, Sub(sender, e)
                                     Dim t = DirectCast(sender, TabControl)
                                     Dim page = t.TabPages(e.Index)
                                     Dim isSelected = (t.SelectedIndex = e.Index)
                                     
                                     Dim bgColor = If(isSelected, ColorPrimary, Color.FromArgb(150, ColorCard))
                                     Dim textColor = If(isSelected, Color.White, ColorText)
                                     
                                     Using brush As New SolidBrush(bgColor)
                                         e.Graphics.FillRectangle(brush, e.Bounds)
                                     End Using
                                     
                                     TextRenderer.DrawText(e.Graphics, page.Text, FontButton, e.Bounds, textColor,
                                                          TextFormatFlags.HorizontalCenter Or TextFormatFlags.VerticalCenter)
                                     
                                     If isSelected Then
                                         Using pen As New Pen(ColorPrimary, 2)
                                             e.Graphics.DrawLine(pen, e.Bounds.Left, e.Bounds.Bottom - 2, 
                                                                e.Bounds.Right, e.Bounds.Bottom - 2)
                                         End Using
                                     End If
                                 End Sub
    End Sub

    ' ─── HELPER METHODS ───────────────────────────────────────────────────────
    Public Function RoundedRect(bounds As Rectangle, radius As Integer) As GraphicsPath
        Dim path As New GraphicsPath()
        Dim diameter = radius * 2
        path.AddArc(bounds.X, bounds.Y, diameter, diameter, 180, 90)
        path.AddArc(bounds.Right - diameter, bounds.Y, diameter, diameter, 270, 90)
        path.AddArc(bounds.Right - diameter, bounds.Bottom - diameter, diameter, diameter, 0, 90)
        path.AddArc(bounds.X, bounds.Bottom - diameter, diameter, diameter, 90, 90)
        path.CloseFigure()
        Return path
    End Function

    Public Sub SetRoundRegion(ctrl As Control, radius As Integer)
        Dim path As New GraphicsPath()
        Dim diameter = radius * 2
        path.AddArc(0, 0, diameter, diameter, 180, 90)
        path.AddArc(ctrl.Width - diameter, 0, diameter, diameter, 270, 90)
        path.AddArc(ctrl.Width - diameter, ctrl.Height - diameter, diameter, diameter, 0, 90)
        path.AddArc(0, ctrl.Height - diameter, diameter, diameter, 90, 90)
        path.CloseFigure()
        ctrl.Region = New Region(path)
    End Sub

    Public Function AdjustBrightness(c As Color, factor As Double) As Color
        Return Color.FromArgb(
            Math.Min(255, Math.Max(0, CInt(c.R * factor))),
            Math.Min(255, Math.Max(0, CInt(c.G * factor))),
            Math.Min(255, Math.Max(0, CInt(c.B * factor))))
    End Function

    ' ─── Form Dragging Support ────────────────────────────────────────────────
    Public Sub EnableDrag(frm As Form)
        Dim isDragging As Boolean = False
        Dim startPoint As Point = Point.Empty

        AddHandler frm.MouseDown, Sub(s, e)
                                      If e.Button = MouseButtons.Left Then
                                          isDragging = True
                                          startPoint = New Point(e.X, e.Y)
                                      End If
                                  End Sub
        
        AddHandler frm.MouseMove, Sub(s, e)
                                      If isDragging Then
                                          Dim p As Point = frm.PointToScreen(e.Location)
                                          frm.Location = New Point(p.X - startPoint.X, p.Y - startPoint.Y)
                                      End If
                                  End Sub
        
        AddHandler frm.MouseUp, Sub(s, e) isDragging = False End Sub
    End Sub

    ' ─── Draw Card Helper ─────────────────────────────────────────────────────
    Public Sub DrawCard(g As Graphics, bounds As Rectangle, Optional radius As Integer = 12, Optional backgroundColor As Color? = Nothing)
        g.SmoothingMode = SmoothingMode.AntiAlias
        Dim bgColor = If(backgroundColor.HasValue, backgroundColor.Value, Color.FromArgb(220, ColorCard))
        
        Using path = RoundedRect(bounds, radius)
            Using brush As New SolidBrush(bgColor)
                g.FillPath(brush, path)
            End Using
            Using pen As New Pen(ColorBorder, 1)
                g.DrawPath(pen, path)
            End Using
        End Using
    End Sub

End Module
