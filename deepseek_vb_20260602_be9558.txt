Imports System.Data.SqlClient
Imports System.Data
Imports Excel = Microsoft.Office.Interop.Excel

Public Class frmEmployee
    Private connString As String = "Server=DCL-ICT-007\DEVELOPER;Database=ESG;Integrated Security=True"
    Private dtEmployees As New DataTable()

    Private Sub frmEmployee_Load(sender As Object, e As EventArgs) Handles MyBase.Load
        ' Apply theme (with or without background image)
        ' Option 1: No background image
        AppTheme.Apply(Me)
        
        ' Option 2: With background image
        ' Dim bgImage As Image = Image.FromFile("C:\Images\employee_bg.jpg")
        ' AppTheme.Apply(Me, bgImage)
        
        ' Enable form dragging (optional)
        AppTheme.EnableDrag(Me)
        
        ' Load data
        LoadEmployeeData()
        SetupFilters()
    End Sub

    Private Sub LoadEmployeeData()
        Try
            Using conn As New SqlConnection(connString)
                Dim query As String = "SELECT FullEmpNo, NAME, SURNAME, SEX, CATEGORY, DepartmentName, " &
                                     "DATE_JOINED, BDATE, Nationality " &
                                     "FROM PAYROLL.dbo.vw_pay_emp_master WHERE Pay = 1 ORDER BY FullEmpNo"

                Dim da As New SqlDataAdapter(query, conn)
                dtEmployees.Clear()
                da.Fill(dtEmployees)
                PopulateGridView()
                UpdateFilterCounters()
            End Using
        Catch ex As Exception
            MessageBox.Show("Error loading data: " & ex.Message, "Database Error", 
                           MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try
    End Sub

    Private Sub PopulateGridView()
        DataGridViewEmployees.AutoGenerateColumns = False
        DataGridViewEmployees.Columns.Clear()

        ' Add columns
        DataGridViewEmployees.Columns.Add("Emp_No", "Emp No")
        DataGridViewEmployees.Columns("Emp_No").DataPropertyName = "FullEmpNo"
        DataGridViewEmployees.Columns("Emp_No").Width = 120

        DataGridViewEmployees.Columns.Add("Emp_Name", "Employee Name")
        DataGridViewEmployees.Columns("Emp_Name").Width = 200

        DataGridViewEmployees.Columns.Add("Gender", "Gender")
        DataGridViewEmployees.Columns("Gender").Width = 80

        DataGridViewEmployees.Columns.Add("Category", "Category")
        DataGridViewEmployees.Columns("Category").Width = 100

        DataGridViewEmployees.Columns.Add("Department", "Department")
        DataGridViewEmployees.Columns("Department").Width = 150

        DataGridViewEmployees.Columns.Add("DOJ", "Date of Join")
        DataGridViewEmployees.Columns("DOJ").DataPropertyName = "DATE_JOINED"
        DataGridViewEmployees.Columns("DOJ").DefaultCellStyle.Format = "dd/MM/yyyy"
        DataGridViewEmployees.Columns("DOJ").Width = 100

        DataGridViewEmployees.Columns.Add("DOB", "Date of Birth")
        DataGridViewEmployees.Columns("DOB").DataPropertyName = "BDATE"
        DataGridViewEmployees.Columns("DOB").DefaultCellStyle.Format = "dd/MM/yyyy"
        DataGridViewEmployees.Columns("DOB").Width = 100

        DataGridViewEmployees.Columns.Add("Nationality", "Nationality")
        DataGridViewEmployees.Columns("Nationality").DataPropertyName = "Nationality"
        DataGridViewEmployees.Columns("Nationality").Width = 120

        ' Create display table with calculated fields
        Dim displayTable As New DataTable()
        displayTable.Columns.Add("FullEmpNo", GetType(String))
        displayTable.Columns.Add("EmpName", GetType(String))
        displayTable.Columns.Add("Gender", GetType(String))
        displayTable.Columns.Add("Category", GetType(String))
        displayTable.Columns.Add("DepartmentName", GetType(String))
        displayTable.Columns.Add("DATE_JOINED", GetType(DateTime))
        displayTable.Columns.Add("BDATE", GetType(DateTime))
        displayTable.Columns.Add("Nationality", GetType(String))

        For Each row As DataRow In dtEmployees.Rows
            Dim displayRow As DataRow = displayTable.NewRow()
            displayRow("FullEmpNo") = row("FullEmpNo").ToString()
            displayRow("EmpName") = $"{row("NAME")} {row("SURNAME")}".Trim()
            
            Dim sexValue = If(row("SEX") Is DBNull.Value, Nothing, Convert.ToInt32(row("SEX")))
            displayRow("Gender") = If(sexValue = 1, "Male", If(sexValue = 0, "Female", "N/A"))
            
            displayRow("Category") = row("CATEGORY").ToString()
            displayRow("DepartmentName") = row("DepartmentName").ToString()
            displayRow("DATE_JOINED") = If(row("DATE_JOINED") Is DBNull.Value, DBNull.Value, Convert.ToDateTime(row("DATE_JOINED")))
            displayRow("BDATE") = If(row("BDATE") Is DBNull.Value, DBNull.Value, Convert.ToDateTime(row("BDATE")))
            displayRow("Nationality") = row("Nationality").ToString()
            
            displayTable.Rows.Add(displayRow)
        Next

        DataGridViewEmployees.DataSource = displayTable
        lblRecordCount.Text = displayTable.Rows.Count.ToString()
    End Sub

    Private Sub SetupFilters()
        ' Setup filter comboboxes
        cmbGender.Items.AddRange(New String() {"All", "Male", "Female"})
        cmbGender.SelectedIndex = 0

        ' Get unique values from data
        Dim categories = dtEmployees.AsEnumerable() _
            .Select(Function(r) r("CATEGORY").ToString()) _
            .Where(Function(c) Not String.IsNullOrEmpty(c)) _
            .Distinct().OrderBy(Function(c) c).ToList()
        categories.Insert(0, "All")
        cmbCategory.DataSource = categories

        Dim departments = dtEmployees.AsEnumerable() _
            .Select(Function(r) r("DepartmentName").ToString()) _
            .Where(Function(d) Not String.IsNullOrEmpty(d)) _
            .Distinct().OrderBy(Function(d) d).ToList()
        departments.Insert(0, "All")
        cmbDepartment.DataSource = departments

        Dim nationalities = dtEmployees.AsEnumerable() _
            .Select(Function(r) r("Nationality").ToString()) _
            .Where(Function(n) Not String.IsNullOrEmpty(n)) _
            .Distinct().OrderBy(Function(n) n).ToList()
        nationalities.Insert(0, "All")
        cmbNationality.DataSource = nationalities
    End Sub

    Private Sub UpdateFilterCounters()
        lblGenderCount.Text = $"({cmbGender.Items.Count - 1})"
        lblCategoryCount.Text = $"({cmbCategory.Items.Count - 1})"
        lblDeptCount.Text = $"({cmbDepartment.Items.Count - 1})"
        lblNationalityCount.Text = $"({cmbNationality.Items.Count - 1})"
    End Sub

    Private Sub ApplyFilters(sender As Object, e As EventArgs) Handles cmbGender.SelectedIndexChanged,
        cmbCategory.SelectedIndexChanged, cmbDepartment.SelectedIndexChanged, cmbNationality.SelectedIndexChanged
        Try
            Dim dv As New DataView(dtEmployees)
            Dim filters As New List(Of String)()

            If cmbGender.SelectedItem?.ToString() <> "All" Then
                Dim genderValue = If(cmbGender.SelectedItem.ToString() = "Male", 1, 0)
                filters.Add($"SEX = {genderValue}")
            End If

            If cmbCategory.SelectedItem?.ToString() <> "All" Then
                filters.Add($"CATEGORY = '{cmbCategory.SelectedItem.ToString().Replace("'", "''")}'")
            End If

            If cmbDepartment.SelectedItem?.ToString() <> "All" Then
                filters.Add($"DepartmentName = '{cmbDepartment.SelectedItem.ToString().Replace("'", "''")}'")
            End If

            If cmbNationality.SelectedItem?.ToString() <> "All" Then
                filters.Add($"Nationality = '{cmbNationality.SelectedItem.ToString().Replace("'", "''")}'")
            End If

            dv.RowFilter = If(filters.Count > 0, String.Join(" AND ", filters), "")
            PopulateGridViewWithFilteredData(dv.ToTable())

        Catch ex As Exception
            MessageBox.Show("Error applying filters: " & ex.Message)
        End Try
    End Sub

    Private Sub PopulateGridViewWithFilteredData(filteredTable As DataTable)
        Dim displayTable As New DataTable()
        displayTable.Columns.Add("FullEmpNo", GetType(String))
        displayTable.Columns.Add("EmpName", GetType(String))
        displayTable.Columns.Add("Gender", GetType(String))
        displayTable.Columns.Add("Category", GetType(String))
        displayTable.Columns.Add("DepartmentName", GetType(String))
        displayTable.Columns.Add("DATE_JOINED", GetType(DateTime))
        displayTable.Columns.Add("BDATE", GetType(DateTime))
        displayTable.Columns.Add("Nationality", GetType(String))

        For Each row As DataRow In filteredTable.Rows
            Dim displayRow As DataRow = displayTable.NewRow()
            displayRow("FullEmpNo") = row("FullEmpNo").ToString()
            displayRow("EmpName") = $"{row("NAME")} {row("SURNAME")}".Trim()
            
            Dim sexValue = If(row("SEX") Is DBNull.Value, Nothing, Convert.ToInt32(row("SEX")))
            displayRow("Gender") = If(sexValue = 1, "Male", If(sexValue = 0, "Female", "N/A"))
            
            displayRow("Category") = row("CATEGORY").ToString()
            displayRow("DepartmentName") = row("DepartmentName").ToString()
            displayRow("DATE_JOINED") = If(row("DATE_JOINED") Is DBNull.Value, DBNull.Value, Convert.ToDateTime(row("DATE_JOINED")))
            displayRow("BDATE") = If(row("BDATE") Is DBNull.Value, DBNull.Value, Convert.ToDateTime(row("BDATE")))
            displayRow("Nationality") = row("Nationality").ToString()
            
            displayTable.Rows.Add(displayRow)
        Next

        DataGridViewEmployees.DataSource = displayTable
        lblRecordCount.Text = displayTable.Rows.Count.ToString()
    End Sub

    Private Sub btnClearFilters_Click(sender As Object, e As EventArgs) Handles btnClearFilters.Click
        cmbGender.SelectedIndex = 0
        cmbCategory.SelectedIndex = 0
        cmbDepartment.SelectedIndex = 0
        cmbNationality.SelectedIndex = 0
        txtEmpID.Clear()
        LoadEmployeeData()
    End Sub

    Private Sub btnSearch_Click(sender As Object, e As EventArgs) Handles btnSearch.Click
        If String.IsNullOrWhiteSpace(txtEmpID.Text) Then
            MessageBox.Show("Please enter Employee ID to search", "Search", 
                           MessageBoxButtons.OK, MessageBoxIcon.Information)
            Return
        End If

        For Each row As DataGridViewRow In DataGridViewEmployees.Rows
            If row.Cells("Emp_No").Value.ToString().ToLower().Contains(txtEmpID.Text.ToLower()) Then
                row.Selected = True
                DataGridViewEmployees.FirstDisplayedScrollingRowIndex = row.Index
                Exit For
            End If
        Next
    End Sub

    Private Sub btnExportExcel_Click(sender As Object, e As EventArgs) Handles btnExportExcel.Click
        Try
            Dim saveDialog As New SaveFileDialog()
            saveDialog.Filter = "Excel Files|*.xlsx"
            saveDialog.Title = "Export Employee Data"
            saveDialog.FileName = $"Employee_Export_{DateTime.Now:yyyyMMdd_HHmmss}.xlsx"

            If saveDialog.ShowDialog() = DialogResult.OK Then
                Dim excelApp As New Excel.Application()
                Dim workbook As Excel.Workbook = excelApp.Workbooks.Add()
                Dim worksheet As Excel.Worksheet = workbook.Sheets(1)

                ' Headers
                For col As Integer = 0 To DataGridViewEmployees.Columns.Count - 1
                    worksheet.Cells(1, col + 1) = DataGridViewEmployees.Columns(col).HeaderText
                Next

                ' Data
                For row As Integer = 0 To DataGridViewEmployees.Rows.Count - 1
                    For col As Integer = 0 To DataGridViewEmployees.Columns.Count - 1
                        worksheet.Cells(row + 2, col + 1) = DataGridViewEmployees.Rows(row).Cells(col).Value?.ToString() ?? ""
                    Next
                Next

                worksheet.Columns.AutoFit()
                workbook.SaveAs(saveDialog.FileName)
                workbook.Close()
                excelApp.Quit()

                Marshal.ReleaseComObject(excelApp)
                
                MessageBox.Show("Data exported successfully!", "Export Complete", 
                               MessageBoxButtons.OK, MessageBoxIcon.Information)
            End If
        Catch ex As Exception
            MessageBox.Show("Error exporting to Excel: " & ex.Message, "Export Error", 
                           MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try
    End Sub
End Class
